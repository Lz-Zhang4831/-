function getAIDirection() {
    const head = snake[0];
    const obstacles = new Set(snake.map(seg => `${seg.x},${seg.y}`));
    // 排除蛇头？BFS时不能经过蛇身（除了蛇头），但蛇头是起点，所以障碍是蛇身除去头。
    // 我们构建障碍集合：所有蛇身（包括头，但起点除外）
    const obstaclesSet = new Set();
    for (let i = 1; i < snake.length; i++) {
        obstaclesSet.add(`${snake[i].x},${snake[i].y}`);
    }
    // BFS到食物
    const pathToFood = bfs(head, food, obstaclesSet);
    if (pathToFood && pathToFood.length > 0) {
        return pathToFood[0]; // 返回第一步方向
    }
    // 如果食物不可达，则跟随蛇尾（或走向安全区域）
    // 找蛇尾位置
    const tail = snake[snake.length - 1];
    // 但注意，蛇尾下一步会消失，所以可能走向蛇尾是安全的
    // 我们计算到蛇尾的路径（不包括蛇尾？但蛇尾是障碍，但到达时蛇尾会移开，所以可以将蛇尾视为可到达）
    // 我们将蛇尾从障碍中移除（因为下一步蛇尾会移动，所以那个位置可能变为可走）
    const obstaclesWithoutTail = new Set(obstaclesSet);
    // 但蛇尾也是蛇身，但我们在BFS时应该允许到达蛇尾位置，因为蛇尾会移开。
    // 所以BFS目标可以是蛇尾，障碍是除了蛇头外的蛇身（但蛇尾也算障碍，但我们可以特殊处理）
    // 更简单：移除蛇尾的障碍
    obstaclesWithoutTail.delete(`${tail.x},${tail.y}`);
    // 但也要注意，如果蛇头旁边就是蛇尾，可能直接走
    const pathToTail = bfs(head, tail, obstaclesWithoutTail);
    if (pathToTail && pathToTail.length > 0) {
        return pathToTail[0];
    }
    // 如果都不可达，则选择安全方向（如没有碰撞的方向）
    // 简单：遍历四个方向，选择不撞墙不撞身的方向，优先选择距离食物最近的方向
    const directions = ['up', 'down', 'left', 'right'];
    let bestDir = null;
    let bestDist = Infinity;
    for (const dir of directions) {
        const vec = DIR_VECTORS[dir];
        const newX = head.x + vec.dx;
        const newY = head.y + vec.dy;
        if (newX < 0 || newX >= COLS || newY < 0 || newY >= ROWS) continue;
        if (obstaclesSet.has(`${newX},${newY}`)) continue;
        // 计算到食物的曼哈顿距离
        const dist = Math.abs(newX - food.x) + Math.abs(newY - food.y);
        if (dist < bestDist) {
            bestDist = dist;
            bestDir = dir;
        }
    }
    if (bestDir) return bestDir;
    // 无路可走，但不应发生
    return direction; // 保持原方向
}
